cmake_minimum_required(VERSION 3.21)
project(sam3_c_client LANGUAGES C)
find_package(Sam3Native CONFIG REQUIRED COMPONENTS C)
add_executable(sam3_c_client main.c)
target_compile_features(sam3_c_client PRIVATE c_std_11)
target_link_libraries(sam3_c_client PRIVATE sam3::c)
sam3_copy_runtime(sam3_c_client)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_target_properties(sam3_c_client PROPERTIES INSTALL_RPATH "$ORIGIN/../${Sam3Native_RUNTIME_LIBDIR}")
endif()
install(TARGETS sam3_c_client RUNTIME DESTINATION bin)
