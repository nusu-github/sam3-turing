cmake_minimum_required(VERSION 3.21)
project(sam3_image_benchmark LANGUAGES CXX)
find_package(Sam3Native CONFIG REQUIRED COMPONENTS CPP)
find_package(Threads REQUIRED)
option(SAM3_BENCHMARK_CUDA "Collect CUDA allocator statistics" OFF)
add_executable(sam3_vision_positions_benchmark ../tools/vision_positions_benchmark.cpp)
add_executable(sam3_rotary_test ../tests/rotary_test.cpp)
add_executable(sam3_video_mask_cache_test ../tests/video_mask_cache_test.cpp)
add_executable(sam3_video_mask_cache_benchmark ../tools/video_mask_cache_benchmark.cpp)
add_executable(sam3_image_benchmark ../tools/image_benchmark.cpp)
add_executable(sam3_image_results_test ../tests/image_results_test.cpp)
add_executable(sam3_rank_executor_test ../tests/rank_executor_test.cpp)
target_link_libraries(sam3_rank_executor_test PRIVATE Threads::Threads)
add_executable(sam3_video_collective_test ../tests/video_collective_test.cpp)
add_executable(sam3_video_collective_probe ../tools/video_collective_probe.cpp)
add_executable(sam3_video_multidevice_probe ../tools/video_multidevice_probe.cpp)
if(SAM3_BENCHMARK_CUDA)
  target_compile_definitions(sam3_video_mask_cache_benchmark PRIVATE SAM3_BENCH_CUDA)
  add_executable(sam3_rotary_stream_test ../tests/rotary_test.cpp)
  target_compile_definitions(sam3_rotary_stream_test PRIVATE SAM3_TEST_CUDA_STREAM)
  target_link_libraries(sam3_rotary_stream_test PRIVATE sam3::cpp)
  set_target_properties(sam3_rotary_stream_test PROPERTIES
    INSTALL_RPATH "$ORIGIN/../${Sam3Native_RUNTIME_LIBDIR}")
  sam3_copy_runtime(sam3_rotary_stream_test)
  install(TARGETS sam3_rotary_stream_test RUNTIME DESTINATION bin)
  target_compile_definitions(sam3_vision_positions_benchmark PRIVATE SAM3_BENCH_CUDA)
  target_compile_definitions(sam3_image_benchmark PRIVATE SAM3_BENCH_CUDA)
  target_compile_definitions(sam3_image_results_test PRIVATE SAM3_TEST_CUDA_STATS)
endif()
foreach(target sam3_video_mask_cache_test sam3_video_mask_cache_benchmark sam3_rotary_test sam3_vision_positions_benchmark sam3_image_benchmark sam3_image_results_test
               sam3_rank_executor_test sam3_video_collective_test sam3_video_collective_probe sam3_video_multidevice_probe)
  target_link_libraries(${target} PRIVATE sam3::cpp)
  set_target_properties(${target} PROPERTIES
    INSTALL_RPATH "$ORIGIN/../${Sam3Native_RUNTIME_LIBDIR}")
  sam3_copy_runtime(${target})
  install(TARGETS ${target} RUNTIME DESTINATION bin)
endforeach()
