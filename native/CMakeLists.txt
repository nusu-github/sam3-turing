cmake_minimum_required(VERSION 3.18)
project(sam3_native_setup LANGUAGES CXX)
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_executable(sam3_native_smoke smoke/main.cpp)
target_compile_features(sam3_native_smoke PRIVATE cxx_std_17)
target_link_libraries(sam3_native_smoke PRIVATE ${TORCH_LIBRARIES})
if(MSVC)
  file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
  add_custom_command(TARGET sam3_native_smoke POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS}
      $<TARGET_FILE_DIR:sam3_native_smoke>)
endif()
enable_testing()
add_test(NAME native_cpu COMMAND sam3_native_smoke cpu)
option(SAM3_TEST_CUDA "Run the CUDA smoke test (requires a GPU)" OFF)
if(SAM3_TEST_CUDA)
  add_test(NAME native_cuda COMMAND sam3_native_smoke cuda)
endif()
