cmake_minimum_required(VERSION 3.18)
project(sam3_native LANGUAGES CXX)
option(SAM3_WITH_CUDA "Build precompiled CUDA kernels" OFF)
option(SAM3_TEST_CUDA "Run CUDA tests (requires a GPU)" OFF)
if(SAM3_WITH_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures for native kernels")
  endif()
  enable_language(CUDA)
endif()
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
include(GenerateExportHeader)
add_library(sam3_native SHARED src/ops.cpp src/edt.cpp src/components.cpp src/register.cpp)
generate_export_header(sam3_native)
target_include_directories(sam3_native PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(sam3_native PUBLIC cxx_std_17)
target_link_libraries(sam3_native PUBLIC ${TORCH_LIBRARIES})
if(SAM3_WITH_CUDA)
  target_sources(sam3_native PRIVATE src/ops_cuda.cu src/edt_cuda.cu src/components_cuda.cu)
  target_compile_definitions(sam3_native PRIVATE SAM3_WITH_CUDA)
  set_target_properties(sam3_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
add_executable(sam3_native_smoke smoke/main.cpp)
target_compile_features(sam3_native_smoke PRIVATE cxx_std_17)
target_link_libraries(sam3_native_smoke PRIVATE ${TORCH_LIBRARIES})
add_executable(sam3_native_ops_test tests/ops_test.cpp)
target_link_libraries(sam3_native_ops_test PRIVATE sam3_native)
if(MSVC)
  file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
  foreach(target sam3_native sam3_native_smoke sam3_native_ops_test)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:${target}>)
  endforeach()
endif()
enable_testing()
add_test(NAME native_cpu COMMAND sam3_native_smoke cpu)
add_test(NAME ops_cpu COMMAND sam3_native_ops_test cpu)
if(SAM3_TEST_CUDA)
  if(NOT SAM3_WITH_CUDA)
    message(FATAL_ERROR "SAM3_TEST_CUDA requires SAM3_WITH_CUDA")
  endif()
  add_test(NAME native_cuda COMMAND sam3_native_smoke cuda)
  add_test(NAME ops_cuda COMMAND sam3_native_ops_test cuda)
endif()
