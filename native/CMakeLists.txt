cmake_minimum_required(VERSION 3.21)
project(sam3_native VERSION 0.1.0 LANGUAGES C CXX)
option(SAM3_BUILD_DEVELOPMENT_TOOLS "Build regression probes and tests" ON)
include(GNUInstallDirs)
option(SAM3_WITH_CUDA "Build precompiled CUDA kernels" OFF)
option(SAM3_FUSE_VISION_QK "Fuse the vision Q/K rotary kernels" ON)
option(SAM3_FUSE_VISION_NORM "Fuse vision normalization, layout and projection casts" ON)
option(SAM3_FUSE_VISION_POSITION "Compute vision position tables per axis" ON)
option(SAM3_TEST_CUDA "Run CUDA tests (requires a GPU)" OFF)
option(SAM3_EXPERIMENT_INT8_GEMM "Build the external-CUTLASS SM75 INT4 FC2 experiment" OFF)
set(SAM3_CUTLASS_ROOT "" CACHE PATH "External CUTLASS checkout for the INT4 experiment")
if(SAM3_WITH_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures for native kernels")
  endif()
  enable_language(CUDA)
endif()
find_package(Torch REQUIRED)
find_package(ICU 72 REQUIRED COMPONENTS uc i18n)
if(NOT ICU_VERSION VERSION_LESS 75)
  message(FATAL_ERROR "Tokenizer parity requires ICU 72-74 (Unicode 15.x); use ICU 74.2. Revalidate normalization/lowercasing before upgrading.")
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
include(GenerateExportHeader)
include(cmake/NativeSources.cmake)
generate_export_header(sam3_native)
include(cmake/Media.cmake)
if(MSVC)
  set_source_files_properties(src/tracking_preprocess.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
else()
  set_source_files_properties(src/tracking_preprocess.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
endif()
target_include_directories(sam3_native PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(sam3_native PUBLIC $<BUILD_INTERFACE:cxx_std_17>)
target_link_libraries(sam3_native PUBLIC "$<BUILD_INTERFACE:${TORCH_LIBRARIES}>")
target_link_libraries(sam3_native PRIVATE ICU::uc ICU::i18n ZLIB::ZLIB Threads::Threads)
if(SAM3_WITH_CUDA)
  target_sources(sam3_native PRIVATE src/roi_align_cuda.cu src/ops_cuda.cu src/edt_cuda.cu src/components_cuda.cu src/rotary_cuda.cu src/vision_fusion_cuda.cu src/rotary_pair_cuda.cu)
  set_source_files_properties(src/ops_cuda.cu src/vision_position_cuda.cu
    PROPERTIES COMPILE_OPTIONS "--extended-lambda")
  target_sources(sam3_native PRIVATE src/vision_position_cuda.cu tools/approx_kernels.cu tools/approx_boundary.cu tools/pixel_transform.cu)
  target_compile_definitions(sam3_native PRIVATE SAM3_WITH_CUDA)
  set_target_properties(sam3_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
if(SAM3_FUSE_VISION_QK)
  set_property(SOURCE ${sam3_vision_sources} APPEND PROPERTY COMPILE_DEFINITIONS SAM3_FUSE_VISION_QK)
endif()
include(cmake/Experiments.cmake)
if(SAM3_FUSE_VISION_NORM)
  set_property(SOURCE ${sam3_vision_sources} APPEND PROPERTY COMPILE_DEFINITIONS SAM3_FUSE_VISION_NORM)
endif()
if(SAM3_FUSE_VISION_POSITION)
  set_property(SOURCE ${sam3_vision_sources} APPEND PROPERTY COMPILE_DEFINITIONS SAM3_FUSE_VISION_POSITION)
endif()
if(SAM3_BUILD_DEVELOPMENT_TOOLS)
  include(cmake/Development.cmake)
endif()
option(SAM3_WITH_OPENCV "Enable upstream OpenCV video preprocessing" OFF)
target_sources(sam3_native PRIVATE src/video_preprocess.cpp)
if(SAM3_WITH_OPENCV)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
  target_link_libraries(sam3_native PRIVATE ${OpenCV_LIBS})
  set_property(SOURCE src/video_preprocess.cpp APPEND PROPERTY INCLUDE_DIRECTORIES "${OpenCV_INCLUDE_DIRS}")
  set_property(SOURCE src/video_preprocess.cpp APPEND PROPERTY COMPILE_DEFINITIONS SAM3_WITH_OPENCV)
endif()

if(SAM3_BUILD_DEVELOPMENT_TOOLS)
  add_executable(sam3_video_preprocess_c_probe tools/video_preprocess_c_probe.c)
  target_compile_features(sam3_video_preprocess_c_probe PRIVATE c_std_11)
  target_link_libraries(sam3_video_preprocess_c_probe PRIVATE sam3_native)
endif()

include(cmake/InstallSDK.cmake)
