cmake_minimum_required(VERSION 3.18)
project(sam3_native LANGUAGES CXX)
option(SAM3_WITH_CUDA "Build precompiled CUDA kernels" OFF)
option(SAM3_TEST_CUDA "Run CUDA tests (requires a GPU)" OFF)
if(SAM3_WITH_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures for native kernels")
  endif()
  enable_language(CUDA)
endif()
find_package(Torch REQUIRED)
find_package(ICU 72 REQUIRED COMPONENTS uc i18n)
if(NOT ICU_VERSION VERSION_LESS 75)
  message(FATAL_ERROR "Tokenizer parity requires ICU 72-74 (Unicode 15.x); use ICU 74.2. Revalidate normalization/lowercasing before upgrading.")
endif()
find_package(ZLIB REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
include(GenerateExportHeader)
add_library(sam3_native SHARED src/interactive_image.cpp src/interactive_decoder.cpp src/interactive_prompt.cpp src/tokenizer.cpp src/roi_align.cpp src/ops.cpp src/edt.cpp src/components.cpp src/weights.cpp src/image_results.cpp src/grounding.cpp src/detection_heads.cpp src/detector_decoder.cpp src/detector_encoder.cpp src/geometry_encoder.cpp src/text_encoder.cpp src/vision_encoder.cpp src/preprocess.cpp src/register.cpp)
generate_export_header(sam3_native)
target_sources(sam3_native PRIVATE src/video_heads.cpp)
target_sources(sam3_native PRIVATE src/memory_encoder.cpp)
target_sources(sam3_native PRIVATE src/memory_attention.cpp)
target_sources(sam3_native PRIVATE src/temporal_memory.cpp)
target_sources(sam3_native PRIVATE src/multiplex.cpp)
target_sources(sam3_native PRIVATE src/multiplex_decoder.cpp)
target_sources(sam3_native PRIVATE src/multiplex_temporal.cpp)
target_sources(sam3_native PRIVATE src/tracking_frame.cpp)
target_include_directories(sam3_native PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(sam3_native PUBLIC cxx_std_17)
target_link_libraries(sam3_native PUBLIC ${TORCH_LIBRARIES})
target_link_libraries(sam3_native PRIVATE ICU::uc ICU::i18n ZLIB::ZLIB)
if(SAM3_WITH_CUDA)
  target_sources(sam3_native PRIVATE src/roi_align_cuda.cu src/ops_cuda.cu src/edt_cuda.cu src/components_cuda.cu)
  target_compile_definitions(sam3_native PRIVATE SAM3_WITH_CUDA)
  set_target_properties(sam3_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
add_executable(sam3_native_smoke smoke/main.cpp)
target_compile_features(sam3_native_smoke PRIVATE cxx_std_17)
target_link_libraries(sam3_native_smoke PRIVATE ${TORCH_LIBRARIES})
add_executable(sam3_native_ops_test tests/ops_test.cpp)
target_link_libraries(sam3_native_ops_test PRIVATE sam3_native)
add_executable(sam3_native_weights_test tests/weights_test.cpp)
target_link_libraries(sam3_native_weights_test PRIVATE sam3_native)
add_executable(sam3_native_multiplex_test tests/multiplex_test.cpp)
target_link_libraries(sam3_native_multiplex_test PRIVATE sam3_native)
add_executable(sam3_weights tools/weights_cli.cpp)
target_link_libraries(sam3_weights PRIVATE sam3_native)
add_executable(sam3_text tools/text_cli.cpp)
target_link_libraries(sam3_text PRIVATE sam3_native)
add_executable(sam3_tokenize tools/tokenizer_cli.cpp)
target_link_libraries(sam3_tokenize PRIVATE sam3_native)
add_executable(sam3_tokenizer_test tests/tokenizer_test.cpp)
target_link_libraries(sam3_tokenizer_test PRIVATE sam3_native)
add_executable(sam3_image_probe tools/image_probe.cpp)
target_link_libraries(sam3_image_probe PRIVATE sam3_native)
add_executable(sam3_interactive tools/interactive_cli.cpp)
target_link_libraries(sam3_interactive PRIVATE sam3_native)
add_executable(sam3_interactive_image tools/interactive_image_cli.cpp)
target_link_libraries(sam3_interactive_image PRIVATE sam3_native)
add_executable(sam3_video_heads tools/video_heads_cli.cpp)
target_link_libraries(sam3_video_heads PRIVATE sam3_native)
add_executable(sam3_memory tools/memory_cli.cpp)
target_link_libraries(sam3_memory PRIVATE sam3_native)
add_executable(sam3_memory_attention tools/memory_attention_cli.cpp)
target_link_libraries(sam3_memory_attention PRIVATE sam3_native)
add_executable(sam3_temporal tools/temporal_cli.cpp)
target_link_libraries(sam3_temporal PRIVATE sam3_native)
add_executable(sam3_multiplex_propagation tools/multiplex_propagation_cli.cpp)
target_link_libraries(sam3_multiplex_propagation PRIVATE sam3_native)
add_executable(sam3_multiplex_temporal tools/multiplex_temporal_cli.cpp)
target_link_libraries(sam3_multiplex_temporal PRIVATE sam3_native)
add_executable(sam3_tracking_frame tools/tracking_frame_cli.cpp)
target_link_libraries(sam3_tracking_frame PRIVATE sam3_native)
add_executable(sam3_detector_transformer tools/detector_cli.cpp)
target_link_libraries(sam3_detector_transformer PRIVATE sam3_native)
add_executable(sam3_geometry tools/geometry_cli.cpp)
target_link_libraries(sam3_geometry PRIVATE sam3_native)
add_executable(sam3_vision tools/vision_cli.cpp)
target_link_libraries(sam3_vision PRIVATE sam3_native)
if(MSVC)
  file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
  foreach(target sam3_native sam3_native_smoke sam3_native_ops_test sam3_native_weights_test sam3_native_multiplex_test sam3_weights sam3_text sam3_tokenize sam3_tokenizer_test sam3_vision sam3_geometry sam3_detector_transformer sam3_image_probe sam3_interactive sam3_interactive_image sam3_video_heads sam3_memory sam3_memory_attention sam3_temporal sam3_multiplex_propagation sam3_multiplex_temporal sam3_tracking_frame)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:${target}>)
  endforeach()
endif()
enable_testing()
add_test(NAME native_cpu COMMAND sam3_native_smoke cpu)
add_test(NAME ops_cpu COMMAND sam3_native_ops_test cpu)
add_test(NAME weights_cpu COMMAND sam3_native_weights_test cpu)
add_test(NAME multiplex_cpu COMMAND sam3_native_multiplex_test cpu)
add_test(NAME tokenizer_cpu COMMAND sam3_tokenizer_test "${CMAKE_CURRENT_SOURCE_DIR}/../sam3/assets/bpe_simple_vocab_16e6.txt.gz")
if(SAM3_TEST_CUDA)
  if(NOT SAM3_WITH_CUDA)
    message(FATAL_ERROR "SAM3_TEST_CUDA requires SAM3_WITH_CUDA")
  endif()
  add_test(NAME native_cuda COMMAND sam3_native_smoke cuda)
  add_test(NAME ops_cuda COMMAND sam3_native_ops_test cuda)
  add_test(NAME weights_cuda COMMAND sam3_native_weights_test cuda)
  add_test(NAME multiplex_cuda COMMAND sam3_native_multiplex_test cuda fp16)
endif()
